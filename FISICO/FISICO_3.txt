CREATE TABLE CLIENTE (
    ID_CLIENTE INTEGER PRIMARY KEY,
    NOME VARCHAR(255),
    EMAIL VARCHAR(255),
    CPF VARCHAR(11),
    RG VARCHAR(8),
    CIDADE VARCHAR(255),
    ESTADO VARCHAR(255),
    LOGRADOURO VARCHAR(255)
);

CREATE TABLE AGENCIA (
    ID_AGENCIA INTEGER PRIMARY KEY,
    NOME VARCHAR(255),
    ESTADO VARCHAR(255),
    CIDADE VARCHAR(255),
    LOGRADOURO VARCHAR(255)
);

CREATE TABLE CONTA_CORRENTE (
    ID_CONTA INTEGER PRIMARY KEY,
    SALDO NUMERIC(18,2),
    TITULAR VARCHAR(255),
    CREDITO NUMERIC(18,2),
    DT_ABERTURA DATE,
    FK_CLIENTE_ID_CLIENTE INTEGER,
    FK_AGENCIA_ID_AGENCIA INTEGER
);

CREATE TABLE EMPRESTIMO (
    ID_EMPRESTIMO INTEGER PRIMARY KEY,
    VALOR NUMERIC(18,2),
    JUROS NUMERIC(4,2),
    DT_EFETIVACAO DATE,
    DT_ENCERRAMENTO DATE,
    FK_CONTA_CORRENTE_ID_CONTA INTEGER
);
